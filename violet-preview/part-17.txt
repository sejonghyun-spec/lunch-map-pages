   const r=table.rect,w=expandedWidth(r,tab);return `<div class="table-wrap" data-layout-y="${r.documentY}" style="position:absolute;${place(r)}width:${px(w)};height:${px(Math.min(r.height,6000))}">${tableHtml(table,table.i,tab)}</div>`;
  }).join('');
  const scroll=[...grouped.entries()].map(([sc,tables],index)=>{
   const r=sc.rect;
   const visualWidth=expandedWidth(r,tab);
   const content=tables.map(t=>`<div style="position:absolute;top:${px(t.rect.documentY-r.documentY)};left:${px(t.rect.documentX-r.documentX)}">${tableHtml(t,t.i,tab)}</div>`).join('');
   const frameWidth=visualWidth;
   const contentWidth=Math.max(frameWidth,...tables.map(t=>
     Math.max(0,t.rect.documentX-r.documentX)+expandedWidth(t.rect,tab)));
   const contentHeight=Math.max(sc.scrollH,r.height,...tables.map(t=>t.rect.documentY-r.documentY+t.rect.height+16));
   const horizontal=sc.scrollW>sc.clientW+24;
   return `<div class="scroll-region" data-layout-y="${r.documentY}" tabindex="0" aria-label="표의 숨겨진 행과 열을 스크롤하여 보세요" style="${place(r)}width:${px(frameWidth)};height:${px(r.height)}">
    <div style="position:relative;min-width:${px(contentWidth)};min-height:${px(contentHeight)}">${content}</div>
   </div>`;
  }).join('');
  return free+scroll;
 }
 function normalizeSectionSpacing(){
  const surface=$('surface');
  const heads=[...surface.querySelectorAll('.section-head')];
  if(!heads.length)return;
  const items=[...surface.querySelectorAll('.table-wrap[data-layout-y],.scroll-region[data-layout-y],.chart-image[data-layout-y],.note-place[data-layout-y]')];
  const sections=heads.map((head,i)=>({head,index:i,
   originalY:Number(head.dataset.layoutY),items:[]}));
  // Source notes are tied to the same section as their nearby table/chart.
  for(const element of items){
   const sourceY=Number(element.dataset.layoutY);
   let index=0;
   for(let j=1;j<sections.length;j++){
    if(sourceY>=sections[j].originalY-6)index=j;else break;
   }
   sections[index].items.push({element,sourceY});
  }
  let lastBottom=0;
  for(let i=0;i<sections.length;i++){
   const section=sections[i];
   const targetTitleY=i===0?Math.max(section.originalY,244):lastBottom+LAYOUT.tableToHeader;
   const titleEl=section.head;
   titleEl.style.top=px(targetTitleY);
   const titleHeight=Math.max(30,titleEl.getBoundingClientRect().height);
   const regular=section.items.filter(x=>!x.element.classList.contains('note-place'));
   const firstY=regular.length?Math.min(...regular.map(x=>x.sourceY)):section.originalY+titleHeight+LAYOUT.headerToTable;
   const minContentTop=targetTitleY+titleHeight+LAYOUT.headerToTable;
   // Preserve relative spacing between elements in a section, but ensure the
   // first table starts at a consistent distance from its heading.
   const offset=targetTitleY-section.originalY+
    Math.max(0,section.originalY+titleHeight+LAYOUT.headerToTable-firstY);
   let bottom=targetTitleY+titleHeight;
   for(const {element,sourceY} of section.items){
    const isNote=element.classList.contains('note-place');
    const newTop=Math.max(isNote?minContentTop:minContentTop,sourceY+offset);
    element.style.top=px(newTop);
    // Keep notes at a shared right-hand rail for all tables.
    if(isNote){element.style.left=px(LAYOUT.noteLeft);element.style.width=px(LAYOUT.noteWidth);}
    const bounds=element.getBoundingClientRect();
    let height=bounds.height;
    if(element.classList.contains('table-wrap')){
      const table=element.querySelector('table.table-el');
      if(table)height=Math.max(height,table.getBoundingClientRect().height);
    }
    bottom=Math.max(bottom,newTop+height);
   }
   lastBottom=bottom;
  }
  // Absolute-positioned surface must grow with the new section positions.
  surface.style.height=px(Math.max(900,lastBottom+LAYOUT.bottomPadding));
 }

 function renderHtml(tab,report){
  const titleTop=report.index===10?100:84;const navTop=report.index===10?178:162;
  const sections=sectionHtml(tab);
  const contentHeight=Math.max(tab.pageHeight,910,...tab.sections.map(x=>x.rect.documentY+50),...tab.tables.map(x=>x.rect.documentY+Math.min(x.rect.height,900)),...tab.charts.map(x=>x.rect.documentY+x.rect.height),...tab.notes.map(x=>x.rect.documentY+x.rect.height))+70;
  $('surface').style.height=px(contentHeight);
  $('surface').innerHTML=`<h1 class="report-title" style="top:${px(titleTop)}">${escape(periodLabel())} ${escape(report.name)}</h1>
   ${tabsMarkup(report,navTop)}
   <div class="view-notice" style="top:${px(navTop+66)}">PDF 4월 양식을 참고한 계층 레이블 시안 · 9월의 실제 항목명/수치는 원본 캡처에서 확인</div>
   ${sections}${chartHtml(tab)}${notesHtml(tab)}${tablesHtml(tab)}`;
  normalizeSectionSpacing();
 }
